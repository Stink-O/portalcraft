# userflow: the user's own flow under tools/play_clip.sh (driven by tools/test_userflow.sh):
#   1. fresh launch, the floor portal placed before Minecraft links (the exit must resolve by itself)
#   2. `map pc_pit_flings` reload with Minecraft already linked
#   3. Minecraft restarted mid-session (the harness runs play_clip.sh --restart-mc on PHASE3)
# Each phase: the plugin resolves the pinned exit on its own (waituntil haveexit never forces a
# check), the pit portal's partner is redirected to the cliff, the look-down shows Minecraft, and
# Chell falls through into the region.
pcb_assert_soft 1
sv_cheats 1
pc_pit_redirect_mode 2
waituntil spawned 600
pcb_log PHASE1 fresh launch, portals placed before the link
pcb_portal 1 -560 -60 64 0 0 0
wait 20
pcb_portal_floor 0 128 -384 -600
waituntil link 9000
waituntil haveexit 3600
wait 90
pcb_assert_exit
pcb_assert_redirect
noclip
setpos 128 -384 -500
setang 89 0 0
wait 150
pcb_assert_frames
screenshot pcb_u1_down
wait 20
setpos 128 -384 -100
noclip
wait 300
pcb_assert_player_below -13500
screenshot pcb_u1_region
wait 20
pcb_log PHASE2 map reload with Minecraft linked
map pc_pit_flings
wait 60
waituntil spawned 600
sv_cheats 1
pcb_portal 1 -560 -60 64 0 0 0
wait 20
pcb_portal_floor 0 128 -384 -600
waituntil haveexit 1800
wait 90
pcb_assert_exit
pcb_assert_redirect
noclip
setpos 128 -384 -500
setang 89 0 0
wait 150
pcb_assert_frames
screenshot pcb_u2_down
wait 20
setpos 128 -384 -100
noclip
wait 300
pcb_assert_player_below -13500
screenshot pcb_u2_region
wait 20
pcb_log PHASE3 restart Minecraft
pc_reset
waituntil nolink 3600
pcb_log PHASE3 link down
waituntil link 9000
wait 300
waituntil haveexit 3600
pcb_portal 1 -560 -60 64 0 0 0
wait 20
pcb_portal_floor 0 128 -384 -600
wait 90
pcb_assert_exit
pcb_assert_redirect
noclip
setpos 128 -384 -500
setang 89 0 0
wait 150
pcb_assert_frames
screenshot pcb_u3_down
wait 20
setpos 128 -384 -100
noclip
wait 300
pcb_assert_player_below -13500
screenshot pcb_u3_region
wait 20
pcb_run_summary
quit
